/// the same as Funcdata::inputConsumeNeeded, with the consume mask of the output given (the model's, not the cached one)
uintb chk_need(Funcdata *d,PcodeOp *op,int4 slot,uintb outc) {
  Varnode *vn = op->getIn(slot);
  if (vn->isAutoLive()) return ~((uintb)0);
  if (op->isCall()) {
    if (slot == 0 || op->isCallWithoutSpec()) return ~((uintb)0);
    FuncCallSpecs *fc = d->getCallSpecs(op);
    if (fc == (FuncCallSpecs *)0 || fc->isInputLocked() || fc->isInputActive()) return ~((uintb)0);
    uintb val = minimalmask(vn->getNZMask());
    int4 bytesConsumed = fc->getInputBytesConsumed(slot);
    if (bytesConsumed != 0) val &= calc_mask(bytesConsumed);
    return val;
  }
  if (!op->isAssignment()) {
    if (op->code() == CPUI_RETURN && slot > 0) {
      if (d->getFuncProto().isOutputLocked() || d->getActiveOutput() != (ParamActive *)0) return ~((uintb)0);
      if (vn->getSpace()->getType() == IPTR_JOIN) return ~((uintb)0);
      uintb val = minimalmask(vn->getNZMask());
      int4 bytesConsumed = d->getFuncProto().getReturnBytesConsumed();
      if (bytesConsumed != 0) val &= calc_mask(bytesConsumed);
      return val;
    }
    return ~((uintb)0);
  }
  Varnode *outvn = op->getOut();
  uintb o = outvn->isAutoLive() ? ~((uintb)0) : outc;
  uintb res;
  if (!op->getConsumeLocal(slot,o,res)) return 0;
  return res;
}
}
